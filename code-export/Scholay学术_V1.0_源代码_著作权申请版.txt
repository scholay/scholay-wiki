   669			defer ticker.Stop()
   670	
   671			log.Println("Online user snapshot task started")
   672	
   673			// Record initial snapshot
   674			if err := userService.RecordOnlineUserSnapshot(); err != nil {
   675				log.Printf("Failed to record initial online user snapshot: %v", err)
   676			}
   677	
   678			for {
   679				select {
   680				case <-ticker.C:
   681					if err := userService.RecordOnlineUserSnapshot(); err != nil {
   682						log.Printf("Failed to record online user snapshot: %v", err)
   683					}
   684				case <-ctx.Done():
   685					log.Println("Online user snapshot task stopped")
   686					return
   687				}
   688			}
   689		}()
   690	
   691		// Setup graceful shutdown
   692		quit := make(chan os.Signal, 1)
   693		signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
   694

   696		addr := ":" + cfg.Server.Port
   697		log.Printf("Server starting on %s", addr)

   699		go func() {
   700			if err := router.Run(addr); err != nil {
   701				log.Fatalf("Failed to start server: %v", err)
   702			}
   703		}()
   704	
   705		// Wait for interrupt signal
   706		<-quit
   707		log.Println("Shutting down server...")
   708	
   709		// Cancel context to stop background tasks
   710		cancel()
   711	
   712		// Shutdown cache worker pool gracefully

   726	func (a *feedbackUserRepoAdapter) FindByID(id uint) (*feedback.UserInfo, error) {
   727		u, err := a.userRepo.FindByID(id)
   728		if err != nil || u == nil {
   729			return nil, err
   730		}
   731		return &feedback.UserInfo{
   732			ID:        u.ID,
   733			Username:  u.Username,
   734			Email:     u.Email,
   735			Phone:     u.Phone,
   736			AvatarURL: u.AvatarURL,
   737		}, nil
   738	}
   739	
   740	func (a *feedbackUserRepoAdapter) FindByIDs(ids []uint) (map[uint]*feedback.UserInfo, error) {
   741		users, err := a.userRepo.FindByIDs(ids)
   742		if err != nil {
   743			return nil, err
   744		}
   745		result := make(map[uint]*feedback.UserInfo)
   746		for id, u := range users {
   747			result[id] = &feedback.UserInfo{
   748				ID:        u.ID,
   749				Username:  u.Username,
   750				Email:     u.Email,
   751				Phone:     u.Phone,
   752				AvatarURL: u.AvatarURL,
   753			}
   754		}
   755		return result, nil
   756	}

   838	// GetDashboardStatistics handles GET /api/v1/admin/dashboard/statistics
   839	func (h *DashboardHandler) GetDashboardStatistics(c *gin.Context) {
   840		stats := &DashboardStatistics{}
   841		now := time.Now()
   842		today := now.Format("2006-01-02")
   843		date7dAgo := now.AddDate(0, 0, -7).Format("2006-01-02")
   844		date30dAgo := now.AddDate(0, 0, -30).Format("2006-01-02")
   845	
   846		// ============================================
   847		// 用户统计
   848		// ============================================
   849	
   850		// 总用户数
   851		h.db.Table("users").Count(&stats.TotalUsers)
   852	
   853		// 今日新增用户
   854		h.db.Table("users").
   855			Where("DATE(created_at) = ?", today).
   856			Count(&stats.NewUsersToday)
   857	
   858		// 7天新增用户
   859		h.db.Table("users").
   860			Where("DATE(created_at) >= ?", date7dAgo).
   861			Count(&stats.NewUsers7d)
   862	
   863		// 30天新增用户
   864		h.db.Table("users").
   865			Where("DATE(created_at) >= ?", date30dAgo).
   866			Count(&stats.NewUsers30d)
   867	
   868		// 30天活跃用户 (有登录记录)
   869		h.db.Table("user_sessions").
   870			Where("last_active_at >= ?", time.Now().AddDate(0, 0, -30)).
   871			Distinct("user_id").
   872			Count(&stats.ActiveUsers30d)
   873	
   874		// 当前在线用户 (5分钟内活跃)
   875		h.db.Table("user_sessions").
   876			Where("last_active_at >= ?", time.Now().Add(-5*time.Minute)).
   877			Count(&stats.OnlineUsers)
   878	
   879		// ============================================
   880		// 订单统计
   881		// ============================================
   882	
   883		// 总订单数
   884		h.db.Table("orders").Count(&stats.TotalOrders)
   885	
   886		// 待支付订单
   887		h.db.Table("orders").
   888			Where("status = ?", "pending").
   889			Count(&stats.PendingOrders)
   890	
   891		// 已支付订单
   892		h.db.Table("orders").
   893			Where("status = ?", "paid").
   894			Count(&stats.PaidOrders)
   895	
   896		// 总收入
   897		h.db.Table("orders").
   898			Where("status = ?", "paid").
   899			Select("COALESCE(SUM(final_amount), 0)").
   900			Scan(&stats.TotalRevenue)
   901	
   902		// 今日收入
   903		h.db.Table("orders").
   904			Where("status = ? AND DATE(paid_at) = ?", "paid", today).
   905			Select("COALESCE(SUM(final_amount), 0)").
   906			Scan(&stats.RevenueToday)
   907	
   908		// 7天收入
   909		h.db.Table("orders").
   910			Where("status = ? AND DATE(paid_at) >= ?", "paid", date7dAgo).
   911			Select("COALESCE(SUM(final_amount), 0)").
   912			Scan(&stats.Revenue7d)
   913	
   914		// 30天收入
   915		h.db.Table("orders").
   916			Where("status = ? AND DATE(paid_at) >= ?", "paid", date30dAgo).
   917			Select("COALESCE(SUM(final_amount), 0)").
   918			Scan(&stats.Revenue30d)
   919	
   920		// ============================================
   921		// 订阅统计
   922		// ============================================
   923	
   924		// 活跃订阅数 (未过期)
   925		h.db.Table("user_subscriptions").
   926			Where("status = ? AND (expired_at IS NULL OR expired_at > ?)", "active", now).
   927			Count(&stats.ActiveSubscriptions)
   928	
   929		// 免费用户 (订阅计划为免费)
   930		h.db.Table("user_subscriptions").
   931			Joins("JOIN subscription_plans ON user_subscriptions.plan_id = subscription_plans.id").
   932			Where("subscription_plans.price = 0 AND user_subscriptions.status = ?", "active").
   933			Count(&stats.FreeUsers)
   934	
   935		// 付费用户
   936		h.db.Table("user_subscriptions").
   937			Joins("JOIN subscription_plans ON user_subscriptions.plan_id = subscription_plans.id").
   938			Where("subscription_plans.price > 0 AND user_subscriptions.status = ?", "active").
   939			Count(&stats.PaidUsers)

   941		// ============================================
   942		// 优惠券统计
   943		// ============================================

   945		// 总优惠券数
